// api/prix-ad.js export default async function handler(req, res) { const { carburant } = req.query; try { // Utilisation d'un proxy CORS pour récupérer le HTML de la source réelle const targetUrl = 'https://www.carburani.com/'; const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(targetUrl)}`; const response = await fetch(proxyUrl); if (!response.ok) throw new Error('Impossible de contacter le proxy'); const data = await response.json(); const html = data.contents; if (!html) throw new Error('Contenu vide reçu'); // Patterns de recherche dans le contenu HTML réel const patterns = { gazole: /Gazole[^0-9]*([0-9]+[.,][0-9]{2,3})/i, sp95: /SP95[^0-9]*([0-9]+[.,][0-9]{2,3})/i, sp98: /SP98[^0-9]*([0-9]+[.,][0-9]{2,3})/i, e10: /SP95[^0-9]*([0-9]+[.,][0-9]{2,3})/i }; const targetPattern = patterns[carburant] || patterns['gazole']; const match = html.match(targetPattern); if (!match || !match[1]) { throw new Error('Format du prix introuvable dans la page'); } const prixClean = parseFloat(match[1].replace(',', '.')); return res.status(200).json({ ok: true, cheapest: prixClean, station: 'Pas de la Case (Andorre)' }); } catch (error) { return res.status(500).json({ ok: false, error: error.message }); } }
